-- Find NPC Function
local function findTargetNpc()
    local name = NpcBox.Text
    if name == "" then return nil end
    return Workspace:FindFirstChild(name, true)
end

-- Teleport to NPC Function
TpButton.MouseButton1Click:Connect(function()
    local npc = findTargetNpc()
    local char = player.Character
    if npc and char and char:FindFirstChild("HumanoidRootPart") then
        local targetPart = npc:FindFirstChild("PrimaryPart") or npc:FindFirstChild("HumanoidRootPart") or npc:FindFirstChildWhichIsA("BasePart")
        if targetPart then
            char.HumanoidRootPart.CFrame = targetPart.CFrame + Vector3.new(0, 3, 0)
        end
    end
end)

-- Force Remote Prompt Interaction Function
InteractButton.MouseButton1Click:Connect(function()
    local npc = findTargetNpc()
    if npc then
        local prompt = npc:FindFirstChildWhichIsA("ProximityPrompt", true)
        if prompt then
            prompt.MaxActivationDistance = 999999
            prompt.RequiresLineOfSight = false
            if fireproximityprompt then
                fireproximityprompt(prompt)
            else
                prompt:InputHoldBegin()
                task.wait(0.1)
                prompt:InputHoldEnd()
            end
        end
    end
end)
