-- Sol's RNG NPC Remote Interaction Helper Script
local Players = game:GetService("Players")
local Workspace = game:GetService("Workspace")
local player = Players.LocalPlayer

-- Target NPC name (e.g., Mari, Jester, etc.)
local targetNpcName = "Mari" -- Change to the desired NPC name

local function interactWithNPC()
    local character = player.Character
    if not character or not character:FindFirstChild("HumanoidRootPart") then return end
    
    -- Find the NPC spawned in the workspace
    local npc = Workspace:FindFirstChild(targetNpcName)
    
    if npc and npc:FindFirstChild("HumanoidRootPart") then
        local originalCFrame = character.HumanoidRootPart.CFrame
        
        -- 1. Teleport to the NPC's location
        character.HumanoidRootPart.CFrame = npc.HumanoidRootPart.CFrame + Vector3.new(0, 3, 0)
        
        task.wait(0.1)
        
        -- 2. Force trigger the ProximityPrompt if available
        for _, descendant in ipairs(npc:GetDescendants()) do
            if descendant:IsA("ProximityPrompt") then
                fireproximityprompt(descendant)
            end
        end
        
        task.wait(0.2)
        
        -- 3. Return to the original position (Optional)
        character.HumanoidRootPart.CFrame = originalCFrame
        print("Attempted interaction with " .. targetNpcName .. ".")
    else
        warn(targetNpcName .. " NPC is not currently spawned.")
    end
end

interactWithNPC()
