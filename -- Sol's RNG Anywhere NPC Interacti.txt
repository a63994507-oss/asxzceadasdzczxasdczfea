local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local player = Players.LocalPlayer

local function triggerNpcInteraction(npcName)
    local targetRemote = ReplicatedStorage:FindFirstChild(npcName, true)
    
    if targetRemote and targetRemote:IsA("RemoteEvent") then
        targetRemote:FireServer()
        print("Successfully sent remote interaction request to: " .. npcName)
    else
        local npcModel = workspace:FindFirstChild(npcName, true)
        if npcModel then
            local prompt = npcModel:FindFirstChildOfClass("ProximityPrompt")
            if prompt then
                fireproximityprompt(prompt)
                print("Successfully forced proximity prompt for: " .. npcName)
            else
                print("ProximityPrompt not found for this NPC.")
            end
        else
            print("Target not found: " .. npcName)
        end
    end
end
