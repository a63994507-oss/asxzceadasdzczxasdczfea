-- Sol's RNG NPC Remote Interaction / Teleport Cheat Script
local Players = game:GetService("Players")
local Workspace = game:GetService("Workspace")
local player = Players.LocalPlayer

-- Target NPC Name (e.g., Jester, Mari, Merchant, etc.)
local targetNpcName = "Jester"

local function forceInteractNpc()
    local character = player.Character
    if not character or not character:FindFirstChild("HumanoidRootPart") then return end
    local hrp = character.HumanoidRootPart

    -- Find NPC inside Workspace
    local npc = Workspace:FindFirstChild(targetNpcName)
    
    if npc and npc:FindFirstChild("HumanoidRootPart") then
        -- Method 1: Teleport player to the NPC to interact directly
        local oldPos = hrp.CFrame
        hrp.CFrame = npc.HumanoidRootPart.CFrame + Vector3.new(0, 3, 0)
        
        task.wait(0.1)
        
        -- Force trigger ProximityPrompt
        local prompt = npc:FindFirstChildWhichIsA("ProximityPrompt", true)
        if prompt then
            prompt.MaxActivationDistance = math.huge
            fireproximityprompt(prompt)
            print("NPC interaction forced successfully!")
        end
        
        task.wait(0.1)
        -- Return to original position (optional)
        hrp.CFrame = oldPos
    else
        warn("The target NPC is currently not spawned or does not exist in the server.")
    end
end

-- Execute script
forceInteractNpc()
